describe('Notifications', () => {
  beforeEach(() => {
    cy.visit('http://localhost:3000/signin');
    cy.get('#username').type('Heath93');
    cy.get('#password').type('s3cret');
    cy.get('.MuiButtonBase-root').contains('Sign In').click();
    cy.wait(2000);
  });

  it('should display a notification when someone likes your transaction', () => {
    cy.get('[data-test=sidenav-notifications]').click();
    cy.wait(1000);
    cy.get('[data-test=notifications-list]').should('exist');
    cy.get('.MuiListItem-root').first().should('be.visible');
  });
});
