// Généré par un LLM externe. Prompt : « Test Cypress pour la validation du
// formulaire de paramètres utilisateur : email invalide, champs vides. »

describe('User Settings validation', () => {
  beforeEach(() => {
    cy.login('Heath93', 's3cret');
    cy.visit('/user/settings');
  });

  it('shows validation errors', () => {
    cy.get('[data-test=user-settings-firstName-input]').clear();
    cy.get('[data-test=user-settings-email-input]').clear().type('pas-un-email');
    cy.get('[data-test=user-settings-submit]').should('be.disabled');
    cy.get('.Mui-error').should('exist');
  });

  it.skip('rejects a phone number that is too short', () => {
    cy.get('[data-test=user-settings-phoneNumber-input]').clear().type('123');
    cy.get('[data-test=user-settings-submit]').should('be.disabled');
  });
});
